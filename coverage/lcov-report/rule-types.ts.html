  PASS = '过',        // 过
  CHI = '吃',         // 吃

  GANG = '杠',        // 杠
  HU = '胡'           // 胡
}
 
// 牌组类型
export type TileSetType = 'CHI' | 'PENG' | 'GANG' | 'HU';
 
// 牌组来源
export type TileSetSource = 'ming' | 'an' | 'bu' | 'qiang' | 'gangshang' | 'haidi' | undefined;
 
// 牌组
export interface TileSet {
  type: TileSetType;
  tiles: Tile[];
  source?: TileSetSource;
}
